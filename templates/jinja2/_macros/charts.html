{# Hand-rolled inline SVG charts for the back-office (O2): no charting library. `sparkline` plots one 30-point revenue series; `bar_chart` plots orders by status. #} {% macro sparkline(series, width=680, height=120) %} {% set values = series|map(attribute='revenue')|list %} {% set peak = values|max %} {% set peak = peak if peak and peak > 0 else 1 %} {% set step = (width / (series|length - 1)) if series|length > 1 else width %} {% endmacro %} {% macro bar_chart(pipeline, width=680, height=170) %} {% set counts = pipeline|map(attribute='count')|list %} {% set peak = counts|max %} {% set peak = peak if peak and peak > 0 else 1 %} {% set n = pipeline|length %} {% set gap = 8 %} {% set bw = (width - gap * (n - 1)) / n %} {% endmacro %}
